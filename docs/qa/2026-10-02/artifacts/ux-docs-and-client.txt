116: // ── Helpers ────────────────────────────────────────────────────────
117: /** Fail-fast if args can't be JSON-encoded. Mirrors Python SDK 0.1.8 fix. */
118: export function ensureJsonSerializable(value: unknown): void {
119:   try {
120:     JSON.stringify(value);
121:   } catch (e) {
122:     const msg = e instanceof Error ? e.message : String(e);
123:     throw new TypeError(
124:       `oversight arguments must be JSON-serializable. Got: ${msg}. ` +
125:         `Convert Maps/Sets/BigInts/circular refs/class instances to plain ` +
126:         `objects/arrays/strings/numbers/booleans/null before the call.`
127:     );
128:   }
129: }
130: 
131: /** Wire shape of a paginated list response. */
132: interface PageEnvelope<T> {
133:   data: T[];
134:   has_more: boolean;
135:   next_cursor: string | null;
136: }
137: 
138: function toPageResult<T>(page: PageEnvelope<T>): PageResult<T> {
139:   return {
140:     data: page.data,
141:     hasMore: page.has_more,
142:     nextCursor: page.next_cursor,
143:   };
144: }
145: 
146: /**
147:  * Coerce an approval timeout to a whole number of seconds.
148:  *
149:  * The live API (`/v1/approvals` → ApprovalCreate.timeout_seconds) types this
150:  * field as `integer` and returns HTTP 422 on a fractional value. Callers may
151:  * naturally pass a float (e.g. a computed duration), so we round to the
152:  * nearest second with Math.round before sending — 30.4 → 30, 30.5 → 31 — and
153:  * clamp to a minimum of 1 so a sub-second value never becomes 0.
154:  */
155: function toIntTimeoutSeconds(value: number): number {
156:   return Math.max(1, Math.round(value));
157: }

159: function buildQuery(
160:   params: Record<string, string | number | undefined>
161: ): string {
162:   const parts: string[] = [];
163:   for (const [key, value] of Object.entries(params)) {
164:     if (value === undefined) continue;
165:     parts.push(`${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`);
166:   }
167:   return parts.length ? `?${parts.join('&')}` : '';
168: }

170: // ── Client ─────────────────────────────────────────────────────────
171: export class SentinelClient {
172:   private readonly apiKey: string;
173:   private readonly apiUrl: string;
174:   private readonly defaultTimeoutSeconds: number;
175: 
176:   constructor(config: SentinelConfig) {
177:     if (!config.apiKey) {
178:       throw new SentinelConfigError(
179:         'configure({ apiKey }) — apiKey is required'
180:       );
181:     }
182:     this.apiKey = config.apiKey;
183:     this.apiUrl = (config.apiUrl ?? DEFAULT_API_URL).replace(/\/$/, '');
184:     this.defaultTimeoutSeconds = config.timeoutSeconds ?? 300;
185:   }
186: 
187:   // ---- low-level HTTP ----
188:   private async request<T = unknown>(
189:     path: string,
190:     init: RequestInit = {}
191:   ): Promise<T> {
192:     const url = `${this.apiUrl}${path}`;
193:     const r = await fetch(url, {
194:       ...init,
195:       headers: {
196:         'User-Agent': USER_AGENT,
197:         'Content-Type': 'application/json',
198:         Authorization: `Bearer ${this.apiKey}`,
199:         ...(init.headers ?? {}),
200:       },
201:     });
202:     if (!r.ok) {
203:       let detail = '';
204:       try {
205:         const body = (await r.json()) as Record<string, unknown>;
206:         const rawDetail = body['detail'] ?? body['message'] ?? body;
207:         detail =
208:           typeof rawDetail === 'string' ? rawDetail : JSON.stringify(rawDetail);
209:       } catch {
210:         const txt = await r.text().catch(() => '');
211:         detail = txt.slice(0, 500);
212:       }
213:       throw new SentinelAPIError(r.status, detail, url);
214:     }
215:     return (await r.json()) as T;
216:   }
217: 
218:   // ---- approvals ----
219:   async createApproval(opts: {
220:     functionName: string;
221:     arguments: unknown;
222:     riskLevel?: RiskLevel;
223:     approvers?: string[];
224:     /** Whole seconds; a fraction is rounded to the nearest second. */
225:     timeoutSeconds?: number;
226:     idempotencyKey?: string;
227:   }): Promise<ApprovalRecord> {
228:     ensureJsonSerializable(opts.arguments);
229:     return this.request<ApprovalRecord>('/v1/approvals', {
230:       method: 'POST',
231:       headers: opts.idempotencyKey
232:         ? { 'Idempotency-Key': opts.idempotencyKey }
233:         : {},
234:       body: JSON.stringify({
235:         function_name: opts.functionName,
236:         arguments: opts.arguments,
237:         risk_level: opts.riskLevel ?? 'medium',
238:         approvers: opts.approvers ?? [],
239:         // API types timeout_seconds as integer and 422s on a fraction.
240:         timeout_seconds: toIntTimeoutSeconds(
241:           opts.timeoutSeconds ?? this.defaultTimeoutSeconds
242:         ),
243:       }),
244:     });
245:   }
246: 
247:   /**
248:    * List approvals one page at a time (cursor pagination). Always sends
249:    * `limit` (default 50) so the server returns the paginated envelope.
250:    */
251:   async listApprovals(opts?: {
252:     status?: string;
253:     limit?: number;
254:     cursor?: string;
255:   }): Promise<PageResult<ApprovalRecord>> {
256:     const qs = buildQuery({
257:       status: opts?.status,
258:       limit: opts?.limit ?? 50,
259:       cursor: opts?.cursor,
260:     });
261:     const page = await this.request<PageEnvelope<ApprovalRecord>>(
262:       `/v1/approvals${qs}`
263:     );
264:     return toPageResult(page);
265:   }
266: 
267:   /**
268:    * Iterate over all approvals, transparently following nextCursor across
269:    * pages until the server reports hasMore=false.
270:    */
271:   async *iterApprovals(opts?: {
272:     status?: string;
273:     limit?: number;
274:   }): AsyncGenerator<ApprovalRecord> {
275:     let cursor: string | undefined;
276:     while (true) {
277:       const page = await this.listApprovals({ ...opts, cursor });
278:       for (const item of page.data) yield item;
279:       if (!page.hasMore || page.nextCursor === null) return;
280:       cursor = page.nextCursor;
281:     }
282:   }
283: 
284:   async getApproval(actionId: string): Promise<ApprovalRecord> {
285:     return this.request<ApprovalRecord>(
286:       `/v1/approvals/${encodeURIComponent(actionId)}`
287:     );
288:   }
289: 
290:   /**
291:    * Block until the approval is decided or the timeout elapses. Uses the
292:    * server-side long-poll endpoint when available (single RTT per ~30 s
293:    * window via Postgres LISTEN/NOTIFY); falls back to plain polling on older
294:    * servers that don't have /wait.
295:    */
296:   async waitForDecision(
297:     actionId: string,
298:     timeoutSeconds?: number
299:   ): Promise<ApprovalRecord> {
300:     const timeout = timeoutSeconds ?? this.defaultTimeoutSeconds;
301:     const deadline = Date.now() + timeout * 1000;
302:     while (true) {
303:       const remaining = Math.max(
304:         1,
305:         Math.min(30, Math.floor((deadline - Date.now()) / 1000))
306:       );
307:       let data: ApprovalRecord;
308:       try {
309:         data = await this.request<ApprovalRecord>(
310:           `/v1/approvals/${encodeURIComponent(actionId)}/wait?timeout=${remaining}`
311:         );
312:       } catch (e) {
313:         if (e instanceof SentinelAPIError && e.statusCode === 404) {
314:           data = await this.getApproval(actionId);
315:         } else {
316:           throw e;
317:         }
318:       }
319:       const status = data.status ?? data.decision;
320:       if (status === 'approved' || status === 'rejected') return data;
321:       if (Date.now() >= deadline) {
322:         throw new ApprovalTimeout(actionId, timeout);
323:       }
324:     }
325:   }

370:   // ---- the decorator surface ----
371:   /**
372:    * Wrap a function so each call pauses for human approval before executing.
373:    *
374:    *   const safeRefund = client.wrap(
375:    *     { riskLevel: 'critical', approvers: ['alice@acme.com'] },
376:    *     async (chargeId: string) => stripe.refunds.create({ charge: chargeId })
377:    *   );
378:    *
379:    *   await safeRefund('ch_abc123');
380:    *
381:    * On approval the wrapped fn runs with the original arguments and its
382:    * return value flows back to the caller. On rejection → ApprovalRejected.
383:    * On timeout → ApprovalTimeout.
384:    */
385:   wrap<Args extends unknown[], R>(
386:     opts: OversightOptions,
387:     fn: (...args: Args) => Promise<R> | R
388:   ): (...args: Args) => Promise<R> {
389:     const fnName = opts.functionName || fn.name || 'anonymous';
390:     return async (...args: Args): Promise<R> => {
391:       // API requires `arguments` be a JSON object (dict). Two ergonomic
392:       // shapes: if the caller passes exactly one plain-object arg, that
393:       // object IS the arguments (named-style). Otherwise wrap positional
394:       // args under `{ args: [...] }`.
395:       const isPlainObject =
396:         args.length === 1 &&
397:         args[0] !== null &&
398:         typeof args[0] === 'object' &&
399:         !Array.isArray(args[0]);
400:       const callArgs: Record<string, unknown> = isPlainObject
401:         ? (args[0] as Record<string, unknown>)
402:         : { args: args as unknown[] };
403:       const idempotencyKey =
404:         typeof opts.idempotencyKey === 'function'
405:           ? opts.idempotencyKey()
406:           : opts.idempotencyKey;
407:       const approval = await this.createApproval({
408:         functionName: fnName,
409:         arguments: callArgs,
410:         riskLevel: opts.riskLevel,
411:         approvers: opts.approvers,
412:         timeoutSeconds: opts.timeoutSeconds,
413:         idempotencyKey,
414:       });
415:       const decision = await this.waitForDecision(
416:         approval.action_id,
417:         opts.timeoutSeconds
418:       );
419:       if (decision.decision === 'rejected' || decision.status === 'rejected') {
420:         throw new ApprovalRejected(
421:           (decision.reason as string) || 'Approval rejected',
422:           approval.action_id
423:         );
424:       }
425:       return await fn(...args);
426:     };
427:   }
428: }
