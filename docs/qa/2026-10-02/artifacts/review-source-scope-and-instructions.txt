54: export function gated<T extends MastraToolLike>(
55:   tool: T,
56:   opts: MastraGateOptions = {}
57: ): T {
58:   const sentinel = opts.client ?? getClient();
59:   const toolName = opts.functionName ?? tool.id ?? 'mastra_tool';
60:   const originalExecute = tool.execute.bind(tool);
61: 
62:   const wrappedExecute = async (...args: unknown[]): Promise<unknown> => {
63:     // Mastra passes a single object: { context, runtimeContext, ... }.
64:     // The `context` field is the parsed inputSchema — that's what we want
65:     // shown to the human approver.
66:     const first = args[0];
67:     const ctx =
68:       first &&
69:       typeof first === 'object' &&
70:       'context' in (first as Record<string, unknown>)
71:         ? (first as { context: unknown }).context
72:         : first;
73: 
74:     const normalized: Record<string, unknown> =
75:       ctx && typeof ctx === 'object' && !Array.isArray(ctx)
76:         ? (ctx as Record<string, unknown>)
77:         : { input: ctx };
78: 
79:     const approval = await sentinel.createApproval({
80:       functionName: toolName,
81:       arguments: normalized,
82:       riskLevel: opts.riskLevel,
83:       approvers: opts.approvers,
84:       timeoutSeconds: opts.timeoutSeconds,
85:     });
86: 
87:     const decision = await sentinel.waitForDecision(
88:       approval.action_id,
89:       opts.timeoutSeconds
90:     );
91: 
92:     if (decision.decision === 'rejected' || decision.status === 'rejected') {
93:       throw new ApprovalRejected(
94:         (decision.reason as string) || 'Approval rejected by Sentinel',
95:         approval.action_id
96:       );
97:     }
98: 
99:     return originalExecute(...args);
100:   };

116: function normalizeArguments(rawInput: unknown): Record<string, unknown> {
117:   // invoke receives `input` as a JSON string. Parse it; if it decodes to a
118:   // plain object that's the arguments dict. Anything else (array, scalar, or
119:   // unparseable) is wrapped so the Sentinel API always receives an object.
120:   if (typeof rawInput === 'string') {
121:     try {
122:       const parsed: unknown = JSON.parse(rawInput);
123:       if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
124:         return parsed as Record<string, unknown>;
125:       }
126:       return { input: parsed };
127:     } catch {
128:       return { input: rawInput };
129:     }
130:   }
131:   if (rawInput && typeof rawInput === 'object' && !Array.isArray(rawInput)) {
132:     return rawInput as Record<string, unknown>;
133:   }
134:   return { input: rawInput };
135: }
