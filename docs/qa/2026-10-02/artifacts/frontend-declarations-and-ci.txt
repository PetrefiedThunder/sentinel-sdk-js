name: release

# semantic-release reads commit messages on main, derives the next
# version per conventional commits, updates CHANGELOG.md + package.json,
# commits + tags, and triggers the existing publish.yml on the tag.
#
# Why npmPublish=false in .releaserc.json: publish.yml handles npm via
# Trusted Publishing (OIDC). semantic-release just tags; publish.yml ships.

# MANUAL TRIGGER ONLY. Running this workflow creates a version tag, which
# triggers publish.yml to ship to npm (IRREVERSIBLE). It is intentionally
# gated behind workflow_dispatch so merging changes never auto-publishes.
#
# To restore automatic release-on-merge, re-add the push trigger:
#   on:
#     push:
#       branches: [main]
#     workflow_dispatch:
on:
  workflow_dispatch:

permissions:
  contents: write
  issues: write
  pull-requests: write

jobs:
  release:
    runs-on: ubuntu-latest
    # Skip when the most recent commit IS the release commit (avoid loops)
    if: "!contains(github.event.head_commit.message, 'chore(release):')"
    steps:
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0
          persist-credentials: true

      - uses: actions/setup-node@v7
        with:
          node-version: '24'

      - name: Install
        run: npm ci

      - name: Build
        run: npm run build

      - name: Test
        run: npm test

      - name: semantic-release
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        # All plugins (incl. commit-analyzer, release-notes-generator, npm,
        # github) and the conventionalcommits preset are pinned in
        # devDependencies and installed by `npm ci`, so they resolve from
        # the local node_modules. Do NOT use `npx --yes ... -p` here: that
        # installed only a subset into an isolated cache that semantic-release
        # could not resolve, causing "Cannot find module" failures.
        run: npx semantic-release

FILE .github/workflows/security.yml
name: security

# Runs on EVERY pull request (no path filter) + pushes to main, so the secret
# scan and dependency audit gate all changes — not just ones touching certain
# paths. (The job previously lived in the path-filtered types-fresh.yml.)
on:
  pull_request:
  push:
    branches: [main]

jobs:
  security:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
        with:
          fetch-depth: 0   # gitleaks needs full history to scan past commits

      - uses: actions/setup-node@v7
        with:
          node-version: '24'

      - run: npm ci

      # Dependency advisory scan. Day-one policy: continue-on-error so a newly
      # disclosed advisory in an existing/transitive dep does not red-line main.
      - name: npm audit (dependency advisories)
        continue-on-error: true
        run: npm audit --audit-level=high

      # Secret scan (hard gate). Run the gitleaks BINARY, not gitleaks-action:
      # the action calls the GitHub API to post PR summaries and fails with
      # "Resource not accessible by integration" without a permissions block.
      # gitleaks auto-detects .gitleaks.toml at the repo root.
      - name: gitleaks (secret scan)
        run: |
          curl -sSL https://github.com/gitleaks/gitleaks/releases/download/v8.24.3/gitleaks_8.24.3_linux_x64.tar.gz | tar -xz -C /tmp gitleaks
          /tmp/gitleaks git . --redact --no-banner

FILE .github/workflows/publish.yml
name: publish to npm

on:
  push:
    tags:
      - 'v*'

# Required for npm Trusted Publishing (OIDC).
# https://docs.npmjs.com/trusted-publishers
permissions:
  contents: read
  id-token: write

jobs:
  publish:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7

      - uses: actions/setup-node@v7
        with:
          node-version: '24'
          registry-url: 'https://registry.npmjs.org'

      - name: Install dependencies
        run: npm ci

      - name: Build
        run: npm run build

      - name: Run unit tests
        run: npm test

      - name: Publish
        run: npm publish --provenance --access public
        env:
          # No NPM_TOKEN — OIDC handles auth via Trusted Publisher config
          # configured in npm settings.
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN || '' }}

name: types-fresh

# Verify the generated OpenAPI types match what the live API exposes.
# Catches forgotten `npm run gen:types` before customers hit a runtime
# mismatch.

on:
  pull_request:
    paths:
      - 'src/generated/**'
      - 'package.json'
      - 'package-lock.json'
      - '.github/workflows/types-fresh.yml'
  schedule:
    # Daily 13:00 UTC catches "API shipped a field, SDK didn't notice"
    - cron: '0 13 * * *'
  workflow_dispatch:

jobs:
  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: '24'
      - run: npm ci
      - name: Regenerate types
        run: npm run gen:types
      - name: Diff
        run: |
          if ! git diff --exit-code src/generated/api.d.ts; then
            echo "::error::Generated types are stale. Run 'npm run gen:types' locally and commit the diff."
            exit 1
          fi

name: test

# Build + unit tests on every pull request and push to main. Previously the
# suite ran only on tag-publish and push-to-main (release.yml), so PRs could
# merge with failing tests; this gates them before merge. Tests import from
# dist/, so the build must run first.
on:
  pull_request:
  push:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: '24'
      - run: npm ci
      - run: npm run build
      - run: npm test

  typecheck:
    # Strict-tsc lint gate. The repo has no ESLint config and relies on
    # tsconfig `strict` + `noUncheckedIndexedAccess`, so `tsc --noEmit` is
    # the minimal type-check parity gate matching the py/go SDK lint jobs.
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-node@v7
        with:
          node-version: '24'
      - run: npm ci
      - run: npm run typecheck

FILE .github/workflows/sbom.yml
name: sbom

# Software Bill of Materials for each tagged release.
# Enterprise buyers require this for compliance (SOC 2, FedRAMP, BSI C5).
#
# Produces a CycloneDX-format SBOM with every transitive dependency, license,
# and vulnerability surface, then uploads it as a release asset alongside
# the source tarball.

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      tag:
        description: 'Tag to generate SBOM for (defaults to latest)'
        required: false

permissions:
  contents: write   # to attach the SBOM to the release

jobs:
  sbom:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
        with:
          ref: ${{ github.event.inputs.tag || github.ref }}

      - uses: actions/setup-python@v7
        with:
          python-version: '3.11'

      - name: Install syft
        run: |
          curl -sSfL https://raw.githubusercontent.com/anchore/syft/main/install.sh \
            | sh -s -- -b /usr/local/bin

      - name: Generate CycloneDX SBOM
        run: |
          syft . \
            -o cyclonedx-json=sbom-${{ github.ref_name }}.cdx.json \
            -o spdx-json=sbom-${{ github.ref_name }}.spdx.json
          # spdx-json is the format GitHub's Dependency graph ingests;
          # cyclonedx is what enterprise security teams expect

      - name: Attach to release
        if: github.event_name == 'release'
        uses: softprops/action-gh-release@v3
        with:
          files: |
            sbom-${{ github.ref_name }}.cdx.json
            sbom-${{ github.ref_name }}.spdx.json

      - name: Upload as workflow artifact (dispatch case)
        if: github.event_name == 'workflow_dispatch'
        uses: actions/upload-artifact@v7
        with:
          name: sbom-${{ github.ref_name }}
          path: |
            sbom-*.cdx.json
            sbom-*.spdx.json
          retention-days: 90
