371:   /**

373:    *

390:     return async (...args: Args): Promise<R> => {

392:       // shapes: if the caller passes exactly one plain-object arg, that
393:       // object IS the arguments (named-style). Otherwise wrap positional
394:       // args under `{ args: [...] }`.
395:       const isPlainObject =
396:         args.length === 1 &&
397:         args[0] !== null &&
398:         typeof args[0] === 'object' &&
399:         !Array.isArray(args[0]);
400:       const callArgs: Record<string, unknown> = isPlainObject
401:         ? (args[0] as Record<string, unknown>)
402:         : { args: args as unknown[] };
403:       const idempotencyKey =
404:         typeof opts.idempotencyKey === 'function'
405:           ? opts.idempotencyKey()
406:           : opts.idempotencyKey;
407:       const approval = await this.createApproval({
408:         functionName: fnName,
409:         arguments: callArgs,
410:         riskLevel: opts.riskLevel,
411:         approvers: opts.approvers,
412:         timeoutSeconds: opts.timeoutSeconds,
413:         idempotencyKey,
414:       });

416:         approval.action_id,
417:         opts.timeoutSeconds
418:       );
